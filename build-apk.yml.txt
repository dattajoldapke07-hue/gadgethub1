name: Build APK

on:
  workflow_dispatch:
  push:
    branches: [main]

env:
  FLET_CLI_NO_RICH_OUTPUT: 1

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install Flet
        run: pip install "flet[all]==0.28.3"

      - name: Build APK
        run: printf 'y\ny\ny\ny\n' | flet build apk --verbose --product "Gadget Hub" --org com.gadgethub --module-name main.py

      - uses: actions/upload-artifact@v4
        with:
          name: gadgethub-apk
          path: build/apk
